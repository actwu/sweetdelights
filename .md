    <script>
        const tabs = ['about', 'product', 'services', 'contact'];
        let currentIdx = 0;

        function switchTab(tabId) {
            currentIdx = tabs.indexOf(tabId);
            
            tabs.forEach(id => {
                const tabElement = document.getElementById(`tab-${id}`);
                const viewElement = document.getElementById(`view-${id}`);
                
                if (id === tabId) {
                    tabElement.className = "tab-link flex-shrink-0 flex-1 px-5 py-2.5 text-xs font-bold tracking-wider rounded-full bg-kafe-accent text-kafe-dark whitespace-nowrap text-center";
                    viewElement.classList.remove('hidden');
                } else {
                    tabElement.className = "tab-link flex-shrink-0 flex-1 px-5 py-2.5 text-xs font-bold tracking-wider rounded-full text-kafe-light/70 hover:bg-white/10 whitespace-nowrap text-center";
                    viewElement.classList.add('hidden');
                }
            });
        }

        function stepTab(direction) {
            let nextIdx = currentIdx + direction;
            if (nextIdx < 0) nextIdx = tabs.length - 1;
            if (nextIdx >= tabs.length) nextIdx = 0;
            switchTab(tabs[nextIdx]);
        }

        lucide.createIcons();
    </script>
